name: Generar APK

# Se ejecuta solo cada vez que cambiás algo en el repositorio,
# y también a mano desde la pestaña Actions > Generar APK > Run workflow.
on:
  push:
    branches: [main]
  workflow_dispatch:

permissions:
  contents: write

jobs:
  apk:
    runs-on: ubuntu-latest
    timeout-minutes: 40
    steps:
      - name: Traer el código
        uses: actions/checkout@v4

      - name: Instalar Node
        uses: actions/setup-node@v4
        with:
          node-version: 22
          cache: npm

      - name: Instalar Java 21
        uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: 21

      - name: Instalar dependencias
        run: npm ci

      - name: Crear el proyecto Android
        run: npx cap add android

      - name: Generar ícono y pantalla de inicio
        run: npx capacitor-assets generate --android

      - name: Copiar la app al proyecto Android
        run: npx cap sync android

      - name: Compilar el APK
        working-directory: android
        run: |
          chmod +x ./gradlew
          ./gradlew assembleDebug --no-daemon

      - name: Ponerle nombre al APK
        run: cp android/app/build/outputs/apk/debug/app-debug.apk Repositorio-UCA.apk

      - name: Guardar el APK (se baja desde Actions)
        uses: actions/upload-artifact@v4
        with:
          name: Repositorio-UCA-apk
          path: Repositorio-UCA.apk
          retention-days: 90

      - name: Publicar el APK en Releases (se baja directo desde el celular)
        continue-on-error: true
        uses: softprops/action-gh-release@v2
        with:
          tag_name: apk-${{ github.run_number }}
          name: Repositorio UCA (versión ${{ github.run_number }})
          body: Descargá el archivo Repositorio-UCA.apk e instalalo en tu celular.
          files: Repositorio-UCA.apk
          make_latest: true
